function num_(x){ var s = String(x).replace(/,/g,'').trim(); if (s==='') return null; var n = Number(s); return isNaN(n) ? null : n; }
function nowStr_(){ return Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy/MM/dd HH:mm:ss'); }

// 日時文字列を TZ非依存の比較キー(YYYYMMDDhhmmss)にする。日付だけの取り出しも。
function dkey_(s){
  var m=String(s||'').match(/(\d{4})[\/\-](\d{1,2})[\/\-](\d{1,2})(?:[ T](\d{1,2}):(\d{1,2})(?::(\d{1,2}))?)?/);
  if(!m) return null;
  function p(x,n){ x=String(x||'0'); while(x.length<n) x='0'+x; return x; }
  return Number(p(m[1],4)+p(m[2],2)+p(m[3],2)+p(m[4],2)+p(m[5],2)+p(m[6],2));
}
function dparts_(s){ var m=String(s||'').match(/(\d{4})[\/\-](\d{1,2})[\/\-](\d{1,2})/); return m?{y:+m[1],mo:+m[2],d:+m[3]}:null; }

// 商品ID -> 発注状態/納品予定/履歴 を発注履歴から作る（表示文字列で読む＝TZずれ無し）
function histMap_(){
  var m = {};
  try {
    var v = hist_().getDataRange().getDisplayValues(); var h = v[0];
    var ci = { dt:h.indexOf('日時'), id:h.indexOf('商品ID'), kind:h.indexOf('種別'), qty:h.indexOf('数量'), memo:h.indexOf('メモ') };
    for (var r=1; r<v.length; r++){
      var pid = String(v[r][ci.id]||'').trim(); if (!pid) continue;
      var dt = String(v[r][ci.dt]||'').trim();
      (m[pid] = m[pid] || []).push({ dt:dt,
        kind:String(v[r][ci.kind]||'発注').trim(), qty:String(v[r][ci.qty]||'').trim(),
        memo:String(v[r][ci.memo]||'').trim(), k:dkey_(dt) });
    }
  } catch(e){}
  for (var kk in m){ m[kk].sort(function(a,b){ return (a.k||0)-(b.k||0); }); }
  return m;
}
function isRecv_(k){ return k.indexOf('受領')===0 || k.indexOf('入荷')===0 || k.indexOf('入庫')===0; }
function orderState_(evs, lead, lastRecvKey){
  // 発注済み＝最後の発注が、最終受領(商品マスタLastReceived or 履歴の入庫)より後
  evs = evs || [];
  var lastOrderK=null, lastOrderDt='', lastRecv=(lastRecvKey||null);
  for (var i=0;i<evs.length;i++){
    var k=evs[i].kind;
    if (k.indexOf('発注')===0){ if(evs[i].k){ lastOrderK=evs[i].k; lastOrderDt=evs[i].dt; } }
    else if (isRecv_(k)){ if(evs[i].k && (lastRecv===null || evs[i].k>lastRecv)) lastRecv=evs[i].k; }
  }
  var ordered = (lastOrderK!==null) && (lastRecv===null || lastOrderK>lastRecv);
  var eta='';
  if (ordered && lead!==null){
    var p=dparts_(lastOrderDt);
    if (p){ var d=new Date(p.y, p.mo-1, p.d+Number(lead)); eta=d.getFullYear()+'-'+('0'+(d.getMonth()+1)).slice(-2)+'-'+('0'+d.getDate()).slice(-2); }
  }
  var hist=[]; for (var j=Math.max(0,evs.length-8);j<evs.length;j++){ hist.push({dt:evs[j].dt,kind:evs[j].kind,qty:evs[j].qty,memo:evs[j].memo}); }
  hist.reverse();
  return { ordered:ordered, oeta:eta, hist:hist };
}

function readAll_(){
  var vals = sheet_().getDataRange().getDisplayValues();
  var head = vals[0];
  function gi(n){ return head.indexOf(n); }
  var ci = { id:gi('商品ID'), name:gi('商品名'), cat:gi('カテゴリー'), sup:gi('発注先'), freq:gi('発注頻度'),
    w:gi('倉庫数量'), s:gi('店舗在庫'), mn:gi('最低在庫'), lead:gi('納期'), lot:gi('発注ロット'),
    ref:gi('参考URL・連絡先'), lr:gi('LastReceived'), ship:gi('配送状況'), eta:gi('お届け予定'),
    exp:gi('賞味期限'), stk:gi('棚卸数'), batch:gi('ロット'), memo:gi('備考'), shipmemo:gi('配送メモ'), price:gi('単価'), imgurl:gi('画像URL') };
  var hm = histMap_();
  var items = [];
  for (var r=1; r<vals.length; r++){
    var row = vals[r];
    var name = String(row[ci.name]||'').trim();
    if (!name) continue;
    var id = String(row[ci.id]||'').trim();
    var w = num_(row[ci.w]) || 0, s = num_(row[ci.s]) || 0, mn = num_(row[ci.mn]);
    var lead = num_(row[ci.lead]), lot = num_(row[ci.lot]);
    var total = w + s;
    var status = (mn===null) ? '未設定' : (total<mn ? '不足' : (total===mn ? 'ギリギリ' : '十分'));
    var ref = String(row[ci.ref]||'').trim();
    var os = orderState_(hm[id], lead, (ci.lr>=0?dkey_(row[ci.lr]):null));
    items.push({ id:id, name:name, cat:(String(row[ci.cat]||'').trim()||'未分類'),
      sup:(String(row[ci.sup]||'').trim()||'未設定'),
      freq:(ci.freq>=0?String(row[ci.freq]||'').trim():''),
      w:w, s:s, total:total, min:mn, status:status, lead:lead, lot:lot, ref:ref,
      ordered:os.ordered, oeta:os.oeta, hist:os.hist,
      ship:String(row[ci.ship]||'').trim(), eta:String(row[ci.eta]||'').trim(), exp:String(row[ci.exp]||'').trim(),
      memo:(ci.memo>=0?String(row[ci.memo]||'').trim():''), shipmemo:(ci.shipmemo>=0?String(row[ci.shipmemo]||'').trim():''),
      stk:(ci.stk>=0?num_(row[ci.stk]):null), batch:(ci.batch>=0?String(row[ci.batch]||'').trim():''),
      price:(ci.price>=0?num_(row[ci.price]):null),
      img:(ci.imgurl>=0 && /^https?:\/\//i.test(String(row[ci.imgurl]||'').trim()) ? String(row[ci.imgurl]).trim() : '') });
  }
  return items;
}

function findRow_(sh, id){
  var vals = sh.getDataRange().getValues(); var idCol = vals[0].indexOf('商品ID');
  for (var r=1;r<vals.length;r++){ if (String(vals[r][idCol]).trim()===String(id).trim()) return {row:r+1, vals:vals, head:vals[0]}; }
  return null;
}

function doInc_(p){
  if (p.token !== TOKEN) return { ok:false, error:'auth' };
  var field = (p.field==='w') ? '倉庫数量' : ((p.field==='s') ? '店舗在庫' : null);
  if (!field) return { ok:false, error:'field' };
  var delta = Number(p.delta);
  if (!(delta===1 || delta===-1)) return { ok:false, error:'delta' };
  var lock = LockService.getScriptLock(); lock.waitLock(8000);
  try {
    var sh = sheet_(); var f = findRow_(sh, p.id); if (!f) return { ok:false, error:'notfound' };
    var fCol = f.head.indexOf(field);
    var cur = Number(String(f.vals[f.row-1][fCol]).replace(/,/g,'')) || 0;
    var nv = cur + delta; if (nv<0) nv=0;
    sh.getRange(f.row, fCol+1).setValue(nv);
    return { ok:true, id:p.id, field:p.field, value:nv };
  } finally { lock.releaseLock(); }
}

// 値を設定（数値=最低在庫/納期/発注ロット/倉庫/店舗/棚卸数、文字=賞味期限/ロット）
function doSet_(p){
  if (p.token !== TOKEN) return { ok:false, error:'auth' };
  var numMap = { min:'最低在庫', lead:'納期', lot:'発注ロット', w:'倉庫数量', s:'店舗在庫', stk:'棚卸数', price:'単価' };
  var txtMap = { exp:'賞味期限', batch:'ロット', name:'商品名', cat:'カテゴリー', sup:'発注先', freq:'発注頻度',
                 memo:'備考', ref:'参考URL・連絡先', ship:'配送状況', eta:'お届け予定', shipmemo:'配送メモ' };
  var isNum = !!numMap[p.field]; var field = numMap[p.field] || txtMap[p.field];
  if (!field) return { ok:false, error:'field' };
  if (p.field==='name' && !String(p.value==null?'':p.value).trim()) return { ok:false, error:'empty' };
  var val;
  if (isNum){ var v=Number(String(p.value).replace(/,/g,'')); if(isNaN(v)||v<0) return {ok:false,error:'value'}; val=v; }
  else { val = String(p.value==null?'':p.value); }
  var lock = LockService.getScriptLock(); lock.waitLock(8000);
  try {
    var sh = sheet_(); var f = findRow_(sh, p.id); if (!f) return { ok:false, error:'notfound' };
    var fCol = f.head.indexOf(field);
    sh.getRange(f.row, fCol+1).setValue(val);
    return { ok:true, id:p.id, field:p.field, value:val };
  } finally { lock.releaseLock(); }
}

// 発注を記録（発注履歴に1行追加）
function doOrder_(p){
  if (p.token !== TOKEN) return { ok:false, error:'auth' };
  var lock = LockService.getScriptLock(); lock.waitLock(8000);
  try {
    var hs = hist_(); var h = hs.getDataRange().getValues()[0];
    var rowObj = {}; rowObj['記録ID']=Utilities.getUuid().substring(0,8); rowObj['日時']=nowStr_();
    rowObj['商品ID']=String(p.id).trim(); rowObj['種別']='発注';
    rowObj['数量']=(p.qty!==undefined&&p.qty!=='')?p.qty:''; rowObj['担当者']='アプリ'; rowObj['メモ']=p.memo||'';
    var line = h.map(function(n){ return rowObj[n]!==undefined?rowObj[n]:''; });
    hs.appendRow(line);
    // 納品予定の再計算用に納期を取る（新規発注は必ず最終受領より後なので lastRecv=null でよい）
    var sh=sheet_(); var f=findRow_(sh,p.id); var lead=null;
    if (f){ var lc=f.head.indexOf('納期'); lead=num_(f.vals[f.row-1][lc]); }
    var os=orderState_(histMap_()[String(p.id).trim()], lead, null);
    return { ok:true, id:p.id, ordered:os.ordered, oeta:os.oeta };
  } finally { lock.releaseLock(); }
}

// 新規商品を追加（商品ID自動採番 P0xx）
function doAdd_(p){
  if (p.token !== TOKEN) return { ok:false, error:'auth' };
  if (!String(p.name||'').trim()) return { ok:false, error:'name' };
  var lock = LockService.getScriptLock(); lock.waitLock(8000);
  try {
    var sh=sheet_(); var vals=sh.getDataRange().getValues(); var head=vals[0];
    var idCol=head.indexOf('商品ID'); var mx=0;
    for (var r=1;r<vals.length;r++){ var m=String(vals[r][idCol]||'').match(/^P(\d+)$/); if (m){ var n=parseInt(m[1],10); if(n>mx)mx=n; } }
    var nid='P'+('000'+(mx+1)).slice(-3);
    var o={}; o['商品ID']=nid; o['商品名']=String(p.name).trim(); o['カテゴリー']=p.cat||'未分類';
    o['発注先']=p.sup||'その他';
    if (p.w!==undefined&&p.w!=='') o['倉庫数量']=num_(p.w)||0;
    if (p.s!==undefined&&p.s!=='') o['店舗在庫']=num_(p.s)||0;
    if (p.min!==undefined&&p.min!=='') o['最低在庫']=num_(p.min);
    if (p.lead!==undefined&&p.lead!=='') o['納期']=num_(p.lead);
    if (p.lot!==undefined&&p.lot!=='') o['発注ロット']=num_(p.lot);
    sh.appendRow(head.map(function(n){ return o[n]!==undefined?o[n]:''; }));
    return { ok:true, id:nid, name:o['商品名'] };
  } finally { lock.releaseLock(); }
}

function todayStr_(){ return Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy-MM-dd'); }
// 日付セルは文字列("2026-10-19")でも日付型でも来るので、年月日だけ取り出して突き合わせる
function dOnly_(s){
  if (Object.prototype.toString.call(s)==='[object Date]' && !isNaN(s)) return Utilities.formatDate(s,'Asia/Tokyo','yyyy-MM-dd');
  var m=String(s||'').match(/(\d{4})\D(\d{1,2})\D(\d{1,2})/); if(!m) return '';
  return m[1]+'-'+('0'+m[2]).slice(-2)+'-'+('0'+m[3]).slice(-2);
}
function checkedToday_(){
  var sh = chkStateSheet_(); var v = sh.getDataRange().getDisplayValues(); var t = todayStr_(); var seen = {}, res = [];
  for (var r=1; r<v.length; r++){ if (dOnly_(v[r][0])===t){ var id=String(v[r][1]).trim(); if (id && !seen[id]){ seen[id]=1; res.push(id); } } }
  return res;
}

  var on = (String(p.on)==='1' || String(p.on)==='true'); var t = todayStr_();
  var lock = LockService.getScriptLock(); lock.waitLock(10000);
  try {
    var sh = chkStateSheet_(); var v = sh.getDataRange().getDisplayValues();
    if (on){ var exists=false; for (var r=1;r<v.length;r++){ if (dOnly_(v[r][0])===t && String(v[r][1]).trim()===id){ exists=true; break; } }
      if (!exists) sh.appendRow([t, id, String(p.name||''), nowStr_()]); }
    else { for (var r=v.length-1; r>=1; r--){ if (dOnly_(v[r][0])===t && String(v[r][1]).trim()===id) sh.deleteRow(r+1); } }
    return { ok:true };
  } finally { lock.releaseLock(); }
}

// ===== 商品写真のアップロード（端末の写真→ドライブ→画像URL列） =====
function imgFolder_(){
  var name = 'dorayama-stock-img';
  var it = DriveApp.getFoldersByName(name);
  return it.hasNext() ? it.next() : DriveApp.createFolder(name);
}
function doImg_(p){
  if (p.token !== TOKEN) return { ok:false, error:'auth' };
  var id = String(p.id||'').trim(); if (!id) return { ok:false, error:'id' };
  var data = String(p.data||''); var m = data.match(/^data:(image\/[\w.+-]+);base64,([\s\S]+)$/);
  if (!m) return { ok:false, error:'data' };
  var lock = LockService.getScriptLock(); lock.waitLock(25000);
  try {
    var blob = Utilities.newBlob(Utilities.base64Decode(m[2]), m[1], id+'_'+Date.now()+'.jpg');
    var file = imgFolder_().createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    var url = 'https://drive.google.com/thumbnail?id='+file.getId()+'&sz=w900';
    var sh = sheet_(); var f = findRow_(sh, id);
    if (f){ var c = f.head.indexOf('画像URL'); if (c<0){ c = f.head.length; sh.getRange(1, c+1).setValue('画像URL'); }
      sh.getRange(f.row, c+1).setValue(url); }
    return { ok:true, url:url };
  } finally { lock.releaseLock(); }
}
function doPost(e){
  var out;
  try {
    var p = JSON.parse((e && e.postData && e.postData.contents) || '{}');
    if (p.action==='img') out = doImg_(p);
    else out = { ok:false, error:'action' };
  } catch(err){ out = { ok:false, error:String(err) }; }
  return ContentService.createTextOutput(JSON.stringify(out)).setMimeType(ContentService.MimeType.JSON);
}

function doGet(e){
  var out; var a = e.parameter.action;
  try {
    if (a==='inc') out=doInc_(e.parameter);
    else if (a==='set') out=doSet_(e.parameter);
    else if (a==='order') out=doOrder_(e.parameter);
    else if (a==='receive') out=doReceive_(e.parameter);
    else if (a==='add') out=doAdd_(e.parameter);
    else if (a==='delete') out=doDelete_(e.parameter);
    else if (a==='submitcheck') out=doSubmitCheck_(e.parameter);
    else if (a==='checkhist') out={ ok:true, hist:listCheckHist_(60) };
    else if (a==='chkset') out=doChkSet_(e.parameter);
    else out={ ok:true, updated:Utilities.formatDate(new Date(),'Asia/Tokyo','yyyy-MM-dd HH:mm'), items:readAll_(), checked:checkedToday_() };
  } catch(err){ out={ ok:false, error:String(err) }; }
  var txt=JSON.stringify(out); var cb=e.parameter.callback;
  if (cb) return ContentService.createTextOutput(cb+'('+txt+')').setMimeType(ContentService.MimeType.JAVASCRIPT);
  return ContentService.createTextOutput(txt).setMimeType(ContentService.MimeType.JSON);
}
